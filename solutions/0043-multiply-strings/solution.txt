class Solution {
    public String multiply(String num1, String num2) {
        int num=0;
        int num3=0;
        int n1=num1.length();
        int n2=num2.length();
        for(int i=0;i<n1;i++){
        
            int d=(num1.charAt(i))-'0';
            num=num*10+d;
            
        }
        for(int j=0;j<n2;j++){    
        
            int d=(num2.charAt(j))-'0';
            num3=num3*10+d;
              
        }
        return String.valueOf(num*num3);     
    }
}
