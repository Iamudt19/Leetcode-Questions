class Solution {
    public int numWaterBottles(int numBottles, int numExchange) {
        int ans=0;
        int n=numBottles;
        while(n>=numExchange){
            ans+=n-(n%numExchange);//Jo bottle bhari hai unka count
            n=n/numExchange+n%numExchange;//khali bottles
        }
        return ans+n;
    }
}
