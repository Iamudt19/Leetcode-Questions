class Solution {
    public int dominantIndex(int[] nums) {
        int idx=-1;
        int max=Integer.MIN_VALUE;

        for(int i=0;i<nums.length;i++){
            if(nums[i]>max){
                max=nums[i];
                idx=i;
            }
        }
        for(int j=0;j<nums.length;j++){
            if(nums[j]!=max && max<2*nums[j]){
                return -1;
            }
        }
        return idx;   
    }
}
