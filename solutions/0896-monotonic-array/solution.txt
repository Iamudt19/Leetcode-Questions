class Solution {
    public boolean isMonotonic(int[] nums) {
        int c=0;
        int d=0;
        if(nums.length==1){
            return true;
        }
        for(int i=1;i<nums.length;i++){
                if(nums[i-1]<=nums[i]){
                    c++;
                }
                if(nums[i-1]>=nums[i]){
                    d++;
                }
                
            }



        
        if(c==nums.length-1 || d==nums.length-1){
                return true;
            }
        return false;
    }
}
