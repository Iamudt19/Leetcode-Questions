class Solution {
    public long maxAlternatingSum(int[] nums) {
        for(int i=0;i<nums.length;i++){
            nums[i]=nums[i]*nums[i];
        }
        Arrays.sort(nums);
        long d=0;
        long s=0;
        int n=nums.length;
        for(int i=0;i<(n)/2;i++){
            d+=(nums[i]);

        }
        for(int j=((n)/2);j<n;j++){
            s+=(nums[j]);
        }
        return (s-d);
        
    }
}
