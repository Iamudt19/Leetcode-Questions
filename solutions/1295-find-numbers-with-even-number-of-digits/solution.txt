class Solution {
    public int findNumbers(int[] nums) {
        int c=0;
        int dc=0;
        for(int i=0;i<nums.length;i++){
            while(nums[i]>0){
                nums[i]=nums[i]/10;
                dc++;
               }
            if(dc%2==0){
                c++;
            }
            dc=0;
        }
        return c;
    }
}
