class Solution {
    public int reversePairs(int[] nums) {
        int c=0;
        for(int i=0;i<nums.length;i++){
            for(int j=0;j<nums.length;j++){
                if(i<j){
                    if((long)nums[i]>2L*nums[j]){
                        c++;
                    }
                }
            }
        }
        return c;
    }
}
